from app.db.models.task_model import Task, TaskStatus
from app.crud.utilities import add_item, remove_item, get_item

from app.schemas.task_schema import TaskCreate, TaskResponse, TaskUpdate

from sqlalchemy import select
from sqlalchemy.sql import func
"""
    All of the CRUD operations for the Task model. 

    Design choice to break each model into its own file, and the crud operations that can be shared across models into their own file.
"""

def create_task(db, task_create: TaskCreate) -> Task:
    task = Task(
        name=task_create.name,
    )

    return add_item(db, task)

def del_task(db, id: int) -> TaskResponse | None:
    deleted_task = remove_item(db, Task, id, TaskResponse)

    return deleted_task

def get_task(db, id: int) -> TaskResponse | None:
    return get_item(db, Task, id, TaskResponse)

def update_task(db, id: int, edits: TaskUpdate):
    existing_task = db.scalar(
        select(Task).where(Task.id == id)
    )
    if not existing_task:
        return None
    edit_data = edits.model_dump(exclude_unset=True)
    
    ## Actual business logic starts here, validating the edits before the task is updated.
    proposed_status = edit_data.get("status", existing_task.status)
    proposed_time = edit_data.get("time_spent_minutes", existing_task.time_spent_minutes)
    proposed_focus = edit_data.get("focus_level", existing_task.focus_level)

    ## This block is here to validate focus and time if the edit is to complete the task. 
    if proposed_status == TaskStatus.COMPLETED:
        ## The time must be greater than 0 for proper tracking, it's not realistic to have a task with no time spent.
        ## This rule is up to interpretation though, in the event a user forgot to add a task in the past and wants it tracked now, but can discuss this later.
        if proposed_time is None or proposed_time <= 0:
            raise ValueError("Time spent must be greater than 0 when completing a task")

        ## Same as the time, a focus level is required when completing a task.
        if proposed_focus is None:
            raise ValueError("Focus is required when completing a task")

    
    for k, v in edit_data.items():
        setattr(existing_task, k, v)
    
    if (
        proposed_status == TaskStatus.IN_PROGRESS
        and existing_task.started_at is None
    ):
        ## If the task is being started, and it hasn't been started yet, set the started_at time to the current time.
        existing_task.started_at = func.now()

    if (
        proposed_status == TaskStatus.COMPLETED
        and existing_task.completed_at is None
    ):
        ## If the task is being completed, and the complete time has not been set yet, set the completed_at time to the current time.
        ## There is an edge case here to discuss where task is set to completed but it was never set to in progress. First thing that comes to mind is to set the started_at time to the current time if it hasn't been set yet.
        existing_task.completed_at = func.now()
    
    try:
        db.commit()
        db.refresh(existing_task)

        return existing_task

    except Exception:
        db.rollback()
        raise

def list_tasks(db) -> list[Task]:
    tasks = db.scalars(
        select(Task).order_by(Task.created_at.desc())
    ).all()


    return tasks

from sqlalchemy import select


def add_item(db, model):
    try:
        db.add(model)
        db.commit()
        db.refresh(model)

        return model

    except Exception:
        db.rollback()
        raise

def remove_item(db, model, id, response_model=None):
    existing_item = db.scalar(select(model).where(model.id == id))

    if not existing_item:
        return None

    deleted_item = response_model.model_validate(existing_item) if response_model else existing_item

    try:
        db.delete(existing_item)
        db.commit()

        return deleted_item
    except Exception:
        db.rollback()
        raise
    
def get_item(db, model, id, response_model=None):
    item = db.scalar(select(model).where(model.id == id))

    if not item:
        return None

    if response_model:
        return response_model.model_validate(item)
    else:
        return item

from app.crud.utilities import add_item, remove_item, get_item

from sqlalchemy import select
from sqlalchemy.sql import func
from typing import Optional
from datetime import date, timedelta
from app.db.models.weekly_summ_model import WeeklySummary

def save_weekly_summary(db, weekly_summary: WeeklySummary) -> WeeklySummary:
    existing_weekly_summary = db.scalar(select(WeeklySummary).where(WeeklySummary.week_start_date == weekly_summary.week_start_date))

    if existing_weekly_summary:
        existing_weekly_summary.week_end_date = weekly_summary.week_end_date
        existing_weekly_summary.total_tasks_created = weekly_summary.total_tasks_created
        existing_weekly_summary.total_tasks_started = weekly_summary.total_tasks_started
        existing_weekly_summary.total_tasks_completed = weekly_summary.total_tasks_completed
        existing_weekly_summary.total_time_spent_minutes = weekly_summary.total_time_spent_minutes
        existing_weekly_summary.average_focus_level = weekly_summary.average_focus_level
        existing_weekly_summary.category_distribution = weekly_summary.category_distribution
        existing_weekly_summary.weekly_summary_notes = weekly_summary.weekly_summary_notes
        existing_weekly_summary.weekly_summary_suggestions = weekly_summary.weekly_summary_suggestions
        existing_weekly_summary.related_tasks = weekly_summary.related_tasks

        db.commit()
        db.refresh(existing_weekly_summary)

        return existing_weekly_summary
    else:
        return add_item(db, weekly_summary) 

def get_weekly_summary(db, requested_date: date) -> Optional[WeeklySummary]:
    return db.scalar(select(WeeklySummary).where(
        WeeklySummary.week_start_date <= requested_date,
         WeeklySummary.week_end_date >= requested_date
         )
         )

## Went ahead and defined some initial task statuses ahead, open to change based on feedback.
class TaskStatus(str, Enum):
    PLANNED = "planned"
    IN_PROGRESS = "in-progress"
    COMPLETED = "completed"

class Category(str, Enum):
    PERSONAL = "personal"
    WORK = "work"
    LEARNING = "learning"
    FITNESS = "fitness"
    OTHER = "other"
    UNCATEGORIZED = "uncategorized"

class FocusLevel(int, Enum):
    VERY_LOW = 1
    LOW = 2
    MEDIUM = 3
    HIGH = 4
    VERY_HIGH = 5

class Task(Base):
    """Model for the task table.

        Fields:
        - id: int
        - name: str
        - category: str
        - status: str
        - time_spent_minutes: int | None
        - focus_level: int | None
        - created_at: datetime
        - started_at: datetime | None
        - updated_at: datetime
        - completed_at: datetime | None

    """
    __tablename__ = "tasks"

    id: Mapped[int] = mapped_column(
        Integer,
        primary_key=True,
        autoincrement=True,
     )

    name: Mapped[str] = mapped_column(
        String(255),

        )

    status: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
        default=TaskStatus.PLANNED.value ## Similar to the category, I set the default value to the planned enum, but this is open to change based on feedback.
        )

        nullable=True,

    focus_level: Mapped[int | None] = mapped_column(
        Integer,
        nullable=True

        nullable=False
    )

    started_at: Mapped[datetime | None] = mapped_column(
        DateTime,
        nullable=True

from collections.abc import Generator
from sqlalchemy.orm import Session

load_dotenv()

DATABASE_URL = os.environ["DATABASE_URL"]

engine = create_engine(
    DATABASE_URL,
    pool_pre_ping=True,
    connect_args={"check_same_thread": False},
)

SessionLocal = sessionmaker(
    bind=engine,
    autoflush=False,
    autocommit=False,
)

def get_db() -> Generator[Session, None, None]:
    db = SessionLocal()

    try:
        yield db
    finally:
        db.close()

    deps_type=HistoricalSearchDeps,
    output_type=HistoricalSearchResponse,
    instructions="""

        For any question about past productivity, tasks, workload, focus, categories,
        time spent, or patterns, you MUST call search_productivity_history before answering.

        Base your answer only on data returned by that tool.
        Do not invent historical activity.
        Mention relevant week ranges when useful.
        Only include matched_summary_ids that came from the tool results.
        """
        )

@historical_search_agent.tool
def search_productivity_history(
    ctx: RunContext[HistoricalSearchDeps],
    query: str,
    n_results: int = 3,
    ) -> list[WeeklySummaryResponse]:

    summaries = get_ranked_weekly_summaries(ctx.deps.db, query, n_results)

    return [
        WeeklySummaryResponse.model_validate(summary) for summary in summaries
    ]

def answer_historical_search(
    db: Session,
    query:str
) -> HistoricalSearchResponse:
    response = historical_search_agent.run_sync(
        query,
        deps=HistoricalSearchDeps(db=db),
    )
    return response.output

    return f"""
    week: {weekly_summary.week_start_date} to {weekly_summary.week_end_date}

    tasks created: {weekly_summary.total_tasks_created}
    tasks started: {weekly_summary.total_tasks_started}

    total time spent: {weekly_summary.total_time_spent_minutes} minutes
    average focus level: {weekly_summary.average_focus_level}

    weekly summary notes: {weekly_summary.weekly_summary_notes}

def upsert_weekly_summary_vector(weekly_summary: WeeklySummary):
    document = build_weekly_summary_document(weekly_summary)
    collection.upsert(
        ids=[f"weekly-summary-{weekly_summary.id}"],
        documents=[document],
        metadatas=[
            {
                "weekly_summary_id": weekly_summary.id,
                "week_start_date": str(weekly_summary.week_start_date),
                "week_end_date": str(weekly_summary.week_end_date),
            }
        ],
    )

def search_weekly_summaries_vectors(query: str, n_results: int = 3) -> list[dict]:
    results = collection.query(
        query_texts=[query],
        n_results=n_results,
    )

    matches = []

    # This loop trims all the unnecessary data and only returns the weekly summary id, week start date, week end date, and distance.
    for metadata, distance in zip(results["metadatas"][0], results["distances"][0]):
        matches.append({
            "weekly_summary_id": metadata.get("weekly_summary_id", None),
            "week_start_date": metadata.get("week_start_date", None),
            "week_end_date": metadata.get("week_end_date", None),
            "distance": distance,
        })

    return matches

def calc_weekly_metrics(db, week_start_date: datetime):

    ## The route passes a date. Accept a string too, then pin the window to midnight.
    if isinstance(week_start_date, str):
        week_start_date = datetime.strptime(week_start_date, "%Y-%m-%d")
    elif not isinstance(week_start_date, datetime):
        week_start_date = datetime.combine(week_start_date, datetime.min.time())

    week_start_date = week_start_date.replace(
        hour=0,
        minute=0,
        second=0,
        microsecond=0,
    )
    ## Set week end date to be 7 days after start date. Can possibly adjust this later and make weeks be static like Monday-Sunday, that way if a week starts on a Tuesday, the end date is still sunday.
    week_end_date = week_start_date + timedelta(days=6)
    next_week_start_date = week_start_date + timedelta(days=7)

    ## Get all tasks created between start and end date. < next_week_start_date because we want to include the tasks created on the end date, so setting the end date to be the next week start date is exclusive, leaving retrived tasks only in the questioned week.
    created_tasks = db.scalars(
        select(Task).where(Task.created_at >= week_start_date, Task.created_at < next_week_start_date)
    ).all()

    ## There is a small edge case here where started_tasks will only contribute time to tasks started during the week provided, however completed tasks can have time overlapping into the next week. If a task was started 2 weeks ago and has had time dedicated to it, but was completed this week with additional time added, the time will be counted for the previous week. This can be fixed with some sort of versioning system or tracking the time spent on a task across multiple weeks, but I don't think that is in scope right now and will accept that tradeoff.
    started_tasks = db.scalars(
        select(Task).where(Task.started_at >= week_start_date, Task.started_at < next_week_start_date)
    ).all()

    completed_tasks = db.scalars(
        select(Task).where(Task.completed_at >= week_start_date, Task.completed_at < next_week_start_date)
    ).all()

    active_in_progress_tasks = [
        task
        for task in started_tasks
        if task.status == TaskStatus.IN_PROGRESS
    ]

    worked_tasks = active_in_progress_tasks + completed_tasks

    ## Calculate total tasks created, started (meaning set to in-progress), and completed.
    total_tasks_created = len(created_tasks)
    total_tasks_started = len(started_tasks)
    total_tasks_completed = len(completed_tasks)

    serialized_tasks = serialize_tasks(started_tasks + completed_tasks + created_tasks)


    time_spent_in_prog = sum(task.time_spent_minutes for task in started_tasks if task.time_spent_minutes and task.status == TaskStatus.IN_PROGRESS)
    time_spent_completed = sum(task.time_spent_minutes for task in completed_tasks if task.time_spent_minutes and task.status == TaskStatus.COMPLETED)
    total_time_spent_minutes = time_spent_in_prog + time_spent_completed


    focus_levels = [
        task.focus_level for task in completed_tasks if task.focus_level is not None
    ]
    total_focus_level = sum(focus_levels)
    average_focus_level = total_focus_level / len(focus_levels) if focus_levels else None

    return {
        "week_start_date": week_start_date,
        "week_end_date": week_end_date,
        "task_activity": {
            "created": total_tasks_created,
            "started": total_tasks_started,
            "completed": total_tasks_completed,
        },
        "time_spent": {
            "in_progress": time_spent_in_prog,
            "completed": time_spent_completed,
            "total": total_time_spent_minutes,
        },
        "focus_level_completed_tasks": {
            "average": average_focus_level,
        },
        "category_breakdown": {
            "task_counts": calculate_category_breakdown(worked_tasks),
            "avg_completed_focus_level": calculate_avg_focus_level(completed_tasks),
            "total_time_spent": calculate_total_time_spent(worked_tasks),
        },
        "relevant_tasks": serialized_tasks,

    }

def serialize_tasks(tasks: list[Task]) -> list[dict]:
    unique_tasks = {
        task.id: task
        for task in tasks
    }.values()

    serialized_tasks = [
    TaskResponse.model_validate(task).model_dump(
        mode="json",
        exclude={"created_at", "updated_at"},
    )
    for task in unique_tasks
]

    return serialized_tasks

def calculate_category_breakdown(tasks: list[Task]) -> dict:
    category_breakdown = {
            "work": 0,
            "learning": 0,
            "personal": 0,
            "fitness": 0,
            "other": 0,
            "uncategorized": 0,
        }

    for task in tasks:
        category_breakdown[task.category] += 1

    return category_breakdown

def calculate_avg_focus_level(tasks: list[Task]) -> dict:
    avg_focus_level_count = {
        "work": [0,0],
        "learning": [0,0],
        "personal": [0,0],
        "fitness": [0,0],
        "other": [0,0],
        "uncategorized": [0,0],
    }
    for task in tasks:
        if task.focus_level is None:
            continue

        avg_focus_level_count[task.category][0] += task.focus_level
        avg_focus_level_count[task.category][1] += 1

    avg_focus_level = {}

    for category, values in avg_focus_level_count.items():
        total_focus, count = values

        avg_focus_level[category] = (
            total_focus / count
            if count > 0
            else None
        )
    return avg_focus_level

def calculate_total_time_spent(tasks: list[Task]) -> dict:
    total_time_spent = {
        "work": 0,
        "learning": 0,
        "personal": 0,
        "fitness": 0,
        "other": 0,
        "uncategorized": 0,
    }
    for task in tasks:
        total_time_spent[task.category] += task.time_spent_minutes or 0

    return total_time_spent

def generate_weekly_summary(db, weekly_metrics: dict) -> WeeklySummary:
    prompt = """
        Analyze the provided weekly productivity data.

        Generate:
        - summary: one concise paragraph describing productivity, workload,
        completion activity, time distribution, and focus trends.
        - suggestions: 2-4 specific and actionable suggestions based only on
        the provided data.

        Rules:
        - Do not invent activity or context not present in the data.
        - Do not make assumptions about the user's health, priorities, or goals.
        - Tasks with the same id represent the same task and must not be double-counted.
        - A task may appear in relevant_tasks because it was created, started,
        or completed during the selected week.
        - Use the aggregate metrics as the source of truth for weekly totals.
    """
    client = genai.Client(api_key=os.getenv("GENAI_API_KEY"))


    response = client.models.generate_content(
        model='gemini-flash-latest',
        contents=[
            types.Part.from_text(text=prompt),
            types.Part.from_text(text=json.dumps(weekly_metrics, default=str)),
        ],    
        config=types.GenerateContentConfig(
            response_mime_type="application/json",
            response_schema=GeneratedWeeklySummary,
        ),
    )
    generated_weekly_summary = GeneratedWeeklySummary.model_validate_json(response.text)

    weekly_summary = WeeklySummary(
        week_start_date=weekly_metrics["week_start_date"].date(),
        week_end_date=weekly_metrics["week_end_date"].date(),
        total_tasks_created=weekly_metrics["task_activity"]["created"],
        total_tasks_started=weekly_metrics["task_activity"]["started"],
        total_tasks_completed=weekly_metrics["task_activity"]["completed"],
        total_time_spent_minutes=weekly_metrics["time_spent"]["total"],
        average_focus_level=weekly_metrics["focus_level_completed_tasks"]["average"],
        category_distribution=weekly_metrics["category_breakdown"]["task_counts"],
        weekly_summary_notes=generated_weekly_summary.summary,
        weekly_summary_suggestions=generated_weekly_summary.suggestions,
        related_tasks=[
            {
                "id": task["id"],
                "name": task["name"],
                "category": task["category"],
                "status": task["status"],
            }
            for task in weekly_metrics["relevant_tasks"]
        ],
    )


    saved_weekly_summary = save_weekly_summary(db, weekly_summary)
    ## Upsert the weekly summary to the vector store.
    upsert_weekly_summary_vector(saved_weekly_summary)
    print(f"Weekly summary {saved_weekly_summary.id} upserted to vector store.")
    return saved_weekly_summary

def get_ranked_weekly_summaries(db, query: str, n_results: int = 3) -> list[WeeklySummary]:
    results = search_weekly_summaries_vectors(query, n_results)

    summary_ids = [
        match["weekly_summary_id"] for match in results
    ]

    summaries = db.scalars(
        select(WeeklySummary).where(WeeklySummary.id.in_(summary_ids))
    ).all()

    # This is a dictionary of the summaries by id, so we can easily access the summary by id later.
    summaries_by_id = {
        summary.id: summary for summary in summaries
    }

    # This list comprehension returns the summaries in the order of the summary ids by ranking from the vector store.
    return [
        summaries_by_id[summary_id] for summary_id in summary_ids if summary_id in summaries_by_id
    ]

## Backend persistence instead of localStorage

Decision:
Use FastAPI + SQLite.

Why:
The application already requires server-side GenAI and vector-search
functionality, so centralizing persistence avoids creating separate client
and server data models.

Alternatives considered:
- localStorage
- IndexedDB

Tradeoff:
Adds backend setup complexity, but simplifies data ownership and prevents
API credentials from reaching the browser.

## JavaScript instead of TypeScript

## GEN AI HEADS UP ##
    Front End is not my main area of expertise, I am very much a backend engineer who utilizes multiple different LLMs when designing and implementing FE components and design.
    For this challenge I will continue to lean on them for that, but will do less so than normal so I can showcase my ability to actually learn on the fly.

Decision:
Use JavaScript for the React frontend.

Why:
The challenge does not require TypeScript, and JavaScript reduces setup
and implementation overhead for a one-week build. Additionally I have more expereince writing in JS. 

Tradeoff:
TypeScript would provide stronger compile-time guarantees, but the added
type-definition work is not necessary for the scope of this project.

        Status:
            Same idea as category, the default value will go in as Planned enum value. 
        

    ## Weekly Summary model decisions made:
        10-6 - Evening:
            Mulling over more design questions, when generating the weekly summary, I have the option of 
            looking at all tasks based on completed_at time or created_at, this lead me down the rabbit hole of
            what is important signal. Multiple tasks with a planned but never completed status are obviously signal of something, however since tasks can be created into the future, it may be something that was prioritized over. DECISION best concession is to create a new start_at column that tracks when work actually started on a task. 
                    -> Future design possibilities, similar to a TTL, create a scheduled task set X days after task is scheduled to ask if user is ready to start. This can be variable.
## CRUD Decisions
    First main decision, as I do with other projects and apps, I chose to break the CRUD functionality into individual files for each model, and a general utilities file that contains operations that can be shared between models, like the general add/remove.

    Task_crud:
        Main functions are very simple, just pointing toward utilities. The update function is the important one that contains some business logic that validates input data before changing the task object.
## Summary Decisions:
    Historical summaries are currently derived from the task eventt timestamps and current task state. The task state is not versioned so generating an older week after a subsequent task changes can produce small differences. Task history and tracking can solve this, but I considered it outside of scope.

    Each weekly summary is generated once, then is fetched from persisted storage if a user wants to look back at the summary for previous weeks.

// Thin client for the task API. Paths are relative so Vite can proxy them
// to the FastAPI server on port 8000.

async function readError(response) {
  const text = await response.text()
  if (!text) return response.statusText || 'Request failed'

  try {
    const body = JSON.parse(text)
    if (typeof body.detail === 'string') return body.detail
    if (Array.isArray(body.detail)) {
      return body.detail.map((item) => item.msg || JSON.stringify(item)).join(' ')
    }
    return text
  } catch {
    return text
  }
}

async function request(path, options = {}) {
  let response
  try {
    response = await fetch(path, {
      headers: { 'Content-Type': 'application/json', ...options.headers },
      ...options,
    })
  } catch {
    throw new Error('Cannot reach the API. Start the backend on port 8000.')
  }

  if (!response.ok) {
    throw new Error(await readError(response))
  }

  if (response.status === 204) return null
  return response.json()
}

function rangeLabel(days) {
  if (!days.length) return 'No dates selected'
  const start = days[0].toLocaleDateString(undefined, { month: 'short', day: 'numeric' })
  if (days.length === 1) return start
  const end = days[days.length - 1]
  const endText = end.toLocaleDateString(
    undefined,
    end.getMonth() === days[0].getMonth() ? { day: 'numeric' } : { month: 'short', day: 'numeric' },
  )
  return `${start} – ${endText}`
}

function tasksOnDays(tasks, days) {
  const keys = new Set(days.map(toDateKey))
  return tasks.filter((task) => {
    const when = taskDay(task)
    return when && keys.has(toDateKey(when))
  })
}

// The task-list date controls decide which days the chart and list share.
function selectByDate(tasks, filter) {
  if (filter.mode === 'range') {
    const start = parseDateInput(filter.start)
    const end = parseDateInput(filter.end)
    const days = start || end ? eachDay(start || end, end || start) : []
    return { chartDays: days, chartLabel: rangeLabel(days), dateTasks: tasksOnDays(tasks, days) }
  }

  if (filter.mode === 'weekday') {
    const name = WEEKDAYS.find((day) => day.value === filter.weekday)?.label ?? 'Weekday'
    if (filter.weekday == null) {
      return { chartDays: [], chartLabel: 'Pick a weekday', dateTasks: [] }
    }
    const dated = tasks.map(taskDay).filter(Boolean)
    if (!dated.length) {
      return { chartDays: [], chartLabel: name, dateTasks: [] }
    }
    const first = new Date(Math.min(...dated.map((day) => day.getTime())))
    const last = new Date(Math.max(...dated.map((day) => day.getTime())))
    first.setHours(0, 0, 0, 0)
    last.setHours(0, 0, 0, 0)
    while (first.getDay() !== filter.weekday) first.setDate(first.getDate() + 1)
    const days = []
    for (let cursor = new Date(first); cursor <= last && days.length < 52; cursor.setDate(cursor.getDate() + 7)) {
      days.push(new Date(cursor))
    }
    const matched = tasks.filter((task) => taskDay(task)?.getDay() === filter.weekday)
    return { chartDays: days, chartLabel: name, dateTasks: matched }
  }

  const days = weekDays(filter.weekOffset)
  return {
    chartDays: days,
    chartLabel: filter.weekOffset === 0 ? 'This week' : rangeLabel(days),
    dateTasks: tasksOnDays(tasks, days),
  }
}

function averageFocus(tasks) {
  const scored = tasks.filter((task) => task.focus_level != null)
  if (!scored.length) return '—'
  const total = scored.reduce((sum, task) => sum + task.focus_level, 0)
  return (total / scored.length).toFixed(1)
}

import { useEffect, useRef, useState } from 'react'

export default function HistoricalSearch() {
  const [open, setOpen] = useState(false)
  const [query, setQuery] = useState('')
  const [result, setResult] = useState(null)
  const [status, setStatus] = useState('idle')
  const [error, setError] = useState('')
  const inputRef = useRef(null)
  const rootRef = useRef(null)

  useEffect(() => {
    if (open) inputRef.current?.focus()
  }, [open])

  // A click on the page, outside this control, dismisses the open panel.
  useEffect(() => {
    if (!open) return

    function handlePointerDown(event) {
      if (!rootRef.current?.contains(event.target)) {
        setOpen(false)
      }
    }

    document.addEventListener('pointerdown', handlePointerDown)
    return () => document.removeEventListener('pointerdown', handlePointerDown)
  }, [open])

  async function handleSubmit(event) {
    event.preventDefault()
    const trimmed = query.trim()
    if (!trimmed) return

    setStatus('searching')
    setError('')
    try {
      const answer = await searchHistory(trimmed)
      setResult(answer)
      setStatus('ready')
    } catch (err) {
      setStatus('error')
      setError(err.message)
    }
  }

  const matched = Array.isArray(result?.matched_summary_ids) ? result.matched_summary_ids : []

  return (
    <div className="ai-search" ref={rootRef}>
      <div className="ai-search-bar">
        <span>AI assisted search</span>
        <button
          type="button"
          className="ai-search-icon"
          aria-expanded={open}
          aria-controls="ai-search-panel"
          aria-label="AI assisted search"
          onClick={() => setOpen((value) => !value)}
        >
          <svg viewBox="0 0 16 16" aria-hidden="true">
            <circle cx="7" cy="7" r="4.25" fill="none" stroke="currentColor" strokeWidth="1.6" />
            <path d="M10.2 10.2 13.5 13.5" fill="none" stroke="currentColor" strokeWidth="1.6" strokeLinecap="round" />
          </svg>
        </button>
      </div>

      {open && (
        <form id="ai-search-panel" className="ai-search-panel" onSubmit={handleSubmit}>
          <input
            ref={inputRef}
            value={query}
            onChange={(event) => setQuery(event.target.value)}
            placeholder="Which weeks had the most deep work?"
            aria-label="Question"
          />
          <button className="primary" type="submit" disabled={status === 'searching' || !query.trim()}>
            {status === 'searching' ? 'Searching…' : 'Search'}
          </button>

          {error && <p className="form-error">{error}</p>}

          {result && status === 'ready' && (
            <div className="ai-search-answer">
              <p>{result.answer}</p>
              {matched.length > 0 && (
                <p className="history-matches">Matched summaries: {matched.join(', ')}</p>
              )}
            </div>
          )}
        </form>
      )}
    </div>
  )
}

function DateFilter({ dateFilter, onDateFilter }) {
  const { mode, weekOffset, start, end, weekday } = dateFilter
  const week = weekDays(weekOffset)
  const weekText = `${week[0].toLocaleDateString(undefined, { month: 'short', day: 'numeric' })} – ${week[6].toLocaleDateString(undefined, { month: 'short', day: 'numeric' })}`

  function chooseMode(nextMode) {
    onDateFilter((current) => {
      if (nextMode === 'range' && !current.start && !current.end) {
        const days = weekDays(current.weekOffset)
        return { ...current, mode: nextMode, start: toDateKey(days[0]), end: toDateKey(days[6]) }
      }
      return { ...current, mode: nextMode }
    })
  }

  return (
    <div className="date-filter">
      <div className="filters" role="tablist" aria-label="How to choose dates">
        {[
          ['week', 'Week'],
          ['range', 'Dates'],
          ['weekday', 'Weekday'],
        ].map(([value, label]) => (
          <button key={value} type="button" className={mode === value ? 'is-on' : ''} onClick={() => chooseMode(value)}>
            {label}
          </button>
        ))}
      </div>

      {mode === 'week' && (
        <div className="week-nav">
          <button type="button" className="week-arrow" aria-label="Previous week" onClick={() => onDateFilter((current) => ({ ...current, weekOffset: current.weekOffset - 1 }))}>
            ←
          </button>
          <button
            type="button"
            className="week-arrow"
            aria-label="Next week"
            disabled={weekOffset >= 0}
            onClick={() => onDateFilter((current) => ({ ...current, weekOffset: Math.min(current.weekOffset + 1, 0) }))}
          >
            →
          </button>
          <span className="week-range">{weekText}</span>
          <button type="button" disabled={weekOffset === 0} onClick={() => onDateFilter((current) => ({ ...current, weekOffset: 0 }))}>
            This week
          </button>
        </div>
      )}

      {mode === 'range' && (
        <div className="date-inputs">
          <label>
            Start
            <input type="date" value={start} onChange={(event) => onDateFilter((current) => ({ ...current, start: event.target.value }))} />
          </label>
          <label>
            End
            <input type="date" value={end} onChange={(event) => onDateFilter((current) => ({ ...current, end: event.target.value }))} />
          </label>
        </div>
      )}

      {mode === 'weekday' && (
        <div className="filters" role="group" aria-label="Day of week">
          {WEEKDAYS.map((day) => (
            <button
              key={day.value}
              type="button"
              className={weekday === day.value ? 'is-on' : ''}
              aria-pressed={weekday === day.value}
              onClick={() => onDateFilter((current) => ({ ...current, weekday: day.value }))}
            >
              {day.short}
            </button>
          ))}
        </div>
      )}
    </div>
  )
}

function dateLabel(date) {
  return date.toLocaleDateString(undefined, { month: 'short', day: 'numeric' })
}

export default function TrendChart({ tasks, days, label, showAverage = false }) {
  const buckets = days.map((date) => ({
    key: toDateKey(date),
    date,
    minutes: 0,
    focusTotal: 0,
    focusCount: 0,
    completed: 0,
    completedFocusTotal: 0,
    completedFocusCount: 0,
  }))
  const byKey = new Map(buckets.map((day) => [day.key, day]))

  tasks.forEach((task) => {
    const when = taskDay(task)
    if (!when) return
    const bucket = byKey.get(toDateKey(when))
    if (!bucket) return
    bucket.minutes += task.time_spent_minutes || 0
    if (task.focus_level != null) {
      bucket.focusTotal += task.focus_level
      bucket.focusCount += 1
    }
    if (task.status === 'completed') {
      bucket.completed += 1
      if (task.focus_level != null) {
        bucket.completedFocusTotal += task.focus_level
        bucket.completedFocusCount += 1
      }
    }
  })

  // A date only stays on the chart when at least one task was completed that day.
  const plotted = buckets.filter((day) => day.completed > 0)
  const completedFocusCount = plotted.reduce((sum, day) => sum + day.completedFocusCount, 0)
  const averageFocus = completedFocusCount
    ? plotted.reduce((sum, day) => sum + day.completedFocusTotal, 0) / completedFocusCount
    : null
  const maxMinutes = Math.max(...plotted.map((day) => day.minutes), 1)
  const columnCount = plotted.length + (showAverage && averageFocus != null ? 1 : 0)
  const wide = columnCount > 7

  return (
    <section className="panel chart-panel" aria-labelledby="trend-heading">
      <div className="panel-head">
        <div>
          <p className="eyebrow">{label}</p>
          <h2 id="trend-heading">Time logged</h2>
        </div>
        <p className="panel-note">Follows the dates selected on the task list.</p>
      </div>

      {days.length === 0 ? (
        <p className="empty-note">Choose a week, a date range, or a weekday to plot.</p>
      ) : plotted.length === 0 ? (
        <p className="empty-note">No completed tasks in this view.</p>
      ) : (
        <div className={`chart-scroll ${wide ? 'is-scroll' : ''}`}>
          <div
            className="chart"
            style={{ '--cols': columnCount }}
            role="img"
            aria-label={`Minutes logged for ${label}`}
          >
            {plotted.map((day) => {
              const height = Math.max((day.minutes / maxMinutes) * 100, day.minutes ? 8 : 0)
              const average = day.focusCount ? Math.round(day.focusTotal / day.focusCount) : 0
              return (
                <div className="chart-col" key={day.key}>
                  <div className="chart-bar-wrap">
                    <div
                      className={`chart-bar focus-${average || 'none'}`}
                      style={{ height: `${height}%` }}
                      title={`${dateLabel(day.date)}: ${day.minutes} min, ${day.completed} completed`}
                    />
                  </div>
                  <span className="chart-day">{dayLabel(day.date)}</span>
                  <span className="chart-date">{day.date.getDate()}</span>
                </div>
              )
            })}
            {showAverage && averageFocus != null && (
              <div className="chart-col">
                <div className="chart-bar-wrap">
                  <div

// SQLite often returns "YYYY-MM-DD HH:MM:SS". Normalize before parsing.
export function parseApiDate(value) {
  if (!value) return null
  const normalized = typeof value === 'string' ? value.replace(' ', 'T') : value
  const date = new Date(normalized)
  return Number.isNaN(date.getTime()) ? null : date
}

export const WEEKDAYS = [
  { value: 0, short: 'Sun', label: 'Sundays' },
  { value: 1, short: 'Mon', label: 'Mondays' },
  { value: 2, short: 'Tue', label: 'Tuesdays' },
  { value: 3, short: 'Wed', label: 'Wednesdays' },
  { value: 4, short: 'Thu', label: 'Thursdays' },
  { value: 5, short: 'Fri', label: 'Fridays' },
  { value: 6, short: 'Sat', label: 'Saturdays' },
]

export const WEEK_LENGTH = 7

export function startOfWeek(date) {
  const start = new Date(date)
  start.setHours(0, 0, 0, 0)
  start.setDate(start.getDate() - start.getDay())
  return start
}

// Sunday–Saturday dates for the current week, or earlier when weekOffset is negative.
export function weekDays(weekOffset) {
  const today = new Date()
  today.setHours(0, 0, 0, 0)
  const start = startOfWeek(today)
  start.setDate(start.getDate() + weekOffset * WEEK_LENGTH)
  return Array.from({ length: WEEK_LENGTH }, (_, index) => {
    const date = new Date(start)
    date.setDate(start.getDate() + index)
    return date
  })
}

// Same day the chart uses: finished, then started, then logged.
export function taskDay(task) {
  return parseApiDate(task.completed_at) || parseApiDate(task.started_at) || parseApiDate(task.created_at)
}

// Date inputs are local calendar days. Parsing them as UTC shifts the day.
export function parseDateInput(value) {
  if (!value) return null
  const [year, month, day] = value.split('-').map(Number)
  if (!year || !month || !day) return null
  return new Date(year, month - 1, day)
}

export function eachDay(start, end) {
  const first = new Date(start)
  const last = new Date(end)
  first.setHours(0, 0, 0, 0)
  last.setHours(0, 0, 0, 0)
  const [from, to] = first <= last ? [first, last] : [last, first]
  const days = []
  const cursor = new Date(from)
  while (cursor <= to && days.length < 120) {
    days.push(new Date(cursor))
    cursor.setDate(cursor.getDate() + 1)
  }
  return days
}

export function toDateKey(date) {
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')
  return `${year}-${month}-${day}`
}

export function toDateTimeInput(value) {
  const date = parseApiDate(value)
  if (!date) return ''
  const pad = (part) => String(part).padStart(2, '0')
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}`
}

export function formatWhen(value) {
  const date = parseApiDate(value)
  if (!date) return '—'
  return date.toLocaleString(undefined, {
    month: 'short',
    day: 'numeric',
    hour: 'numeric',
    minute: '2-digit',
  })
}

export function formatMinutes(minutes) {
  if (minutes == null) return '—'
  const hours = Math.floor(minutes / 60)
  const rest = minutes % 60
  if (hours === 0) return `${rest}m`
  if (rest === 0) return `${hours}h`
  return `${hours}h ${rest}m`
}
